批量刪除B站關注，精簡急速版

2026-07-09 默認分類 3433 字 5 分钟阅读

在知乎上看到一個困擾我許久的問題，我認爲學習視頻多的地方應該關注知識區UP主之類的，然而關注了1000個不關聯的人。這難免和我内心相悖，於是就找找有什麽地方可以教我一邊如何批量刪除。然後找到了今天這篇帖子：https://zhuanlan.zhihu.com/p/2043616557302076562
但是，我順便照著他的方式做了修改。修改我貼在下面了。

(async () => {
  // 0.3秒速度配置
  const config = {
    clickDelay: 50,       // 瞬间滚动并点击
    confirmDelay: 100,    // 压缩弹窗动画等待
    perUserDelay: 150,    // 结束后只歇 150ms
    pageDelay: 1200,      // 翻页加载留 1.2 秒
    scrollBehavior: "auto"
  };

  // 移除了 CONFIRM_TEXT 打字确认环节，直接进入执行

  window.__stopBiliUnfollow = false;
  window.stopBiliUnfollow = () => {
    window.__stopBiliUnfollow = true;
    console.warn("已请求停止，当前操作结束后会停下。");
  };

  const sleep = (ms) => new Promise(resolve => setTimeout(resolve, ms));
  const clean = (text) => (text || "").replace(/\s+/g, "").trim();

  function visible(el) {
    const rect = el.getBoundingClientRect();
    return rect.width > 0 && rect.height > 0;
  }

  function getFollowButtons() {
    return [...document.querySelectorAll("button, .follow-btn__trigger, [role='button']")]
      .filter(visible)
      .filter(el => ["已关注", "互相关注"].includes(clean(el.textContent)))
      .filter(el => !el.dataset.biliUnfollowTried);
  }

  function clickByText(texts) {
    const btn = [...document.querySelectorAll("button, .be-dropdown-item, .vui_button, [role='button']")]
      .filter(visible)
      .find(el => texts.includes(clean(el.textContent)));

    if (!btn) return false;
    btn.click();
    return true;
  }

  async function unfollowOne(btn) {
    btn.dataset.biliUnfollowTried = "1";
    btn.scrollIntoView({ behavior: config.scrollBehavior, block: "center" });
    await sleep(config.clickDelay);

    btn.click();
    await sleep(config.confirmDelay);

    clickByText(["取消关注"]);
    await sleep(config.confirmDelay);

    clickByText(["确定", "确认"]);
    await sleep(config.perUserDelay);

    return true;
  }

  let totalCount = 0;

  async function processCurrentPage(pageNumber) {
    let pageCount = 0;

    while (!window.__stopBiliUnfollow) {
      const buttons = getFollowButtons();
      if (!buttons.length) break;

      try {
        await unfollowOne(buttons[0]);
        pageCount++;
        totalCount++;
        console.log(`第 ${pageNumber} 页已处理 ${pageCount} 个，总计 ${totalCount} 个。输入 stopBiliUnfollow() 可停止。`);
      } catch (error) {
        console.error("处理单个关注失败，跳过继续：", error);
        await sleep(1000);
      }
    }

    return pageCount;
  }

  function goNextPage() {
    const nextBtn = [...document.querySelectorAll("button, .vui_button, [role='button']")]
      .filter(visible)
      .find(el => {
        const text = clean(el.textContent);
        return text.includes("下一页")
          && !el.disabled
          && el.getAttribute("aria-disabled") !== "true"
          && !el.className.includes("disabled");
      });

    if (!nextBtn) return false;
    nextBtn.click();
    return true;
  }

  let pageNumber = 1;

  console.log("已经绕过确认，开始光速批量取消关注... 输入 stopBiliUnfollow() 可随时停止。");

  while (!window.__stopBiliUnfollow) {
    console.log(`开始处理第 ${pageNumber} 页`);
    await processCurrentPage(pageNumber);

    await sleep(config.pageDelay);

    if (!goNextPage()) {
      console.log("没有找到可用的下一页按钮，任务结束。");
      break;
    }

    pageNumber++;
    await sleep(config.pageDelay);
  }

  console.log(`完成或已停止。共尝试处理 ${totalCount} 个关注。`);
})();
